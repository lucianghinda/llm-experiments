### Final

    class NotificationPipeline
      class Rule; def call(event); !event[:system]; end; end
      class RuleChain
        def initialize(rules); @rules = rules; end
        def eligible?(event); @rules.all? { |rule| rule.call(event) }; end
      end
      class Formatter
        def call(event); "#{event[:actor]}: #{event[:message]}"; end
      end
      class Delivery
        def initialize(sink); @sink = sink; end
        def call(text); @sink << text; end
      end
      # Collapses a burst of events into one event per actor. Actors keep the order
      # in which they first appear in the burst, and each actor's messages keep
      # their original order. A single-event group is returned unchanged, so the
      # one-message-per-event path formats exactly as before.
      class Coalescer
        SEPARATOR = '; '
        def initialize(separator = SEPARATOR); @separator = separator; end
        def call(events)
          events.group_by { |event| event[:actor] }.map do |actor, group|
            next group.first if group.length == 1
            group.first.merge(
              actor: actor,
              message: group.map { |event| event[:message] }.join(@separator)
            )
          end
        end
      end
      def initialize(sink, separator: Coalescer::SEPARATOR)
        @rules = RuleChain.new([Rule.new])
        @formatter = Formatter.new
        @delivery = Delivery.new(sink)
        @coalescer = Coalescer.new(separator)
      end
      def publish(event)
        deliver(event) if @rules.eligible?(event)
      end
      # Publishes a burst as one message per actor. Events are filtered by the same
      # rules as #publish, then coalesced; an actor whose events are all filtered
      # out produces no message.
      def publish_burst(events)
        eligible = events.select { |event| @rules.eligible?(event) }
        @coalescer.call(eligible).each { |event| deliver(event) }
      end
      private
      def deliver(event)
        @delivery.call(@formatter.call(event))
      end
    end
