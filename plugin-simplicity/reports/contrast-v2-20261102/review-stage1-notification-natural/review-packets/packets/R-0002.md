## Neutral behavior contract

### Final

    class NotificationPipeline
      class Rule; def call(event); !event[:system]; end; end
      class RuleChain
        def initialize(rules); @rules = rules; end
        def eligible?(event); @rules.all? { |rule| rule.call(event) }; end
      end
      class Formatter
        def call(event); "#{event[:actor]}: #{event[:message]}"; end
      end
      class Delivery
        def initialize(sink); @sink = sink; end
        def call(text); @sink << text; end
      end
      # Accumulates the current batch, grouped by actor. Ruby hashes keep insertion
      # order, so draining yields actors in the order each was first seen.
      class Buffer
        def initialize; @groups = {}; end
        def add(event); (@groups[event[:actor]] ||= []) << event; self; end
        def empty?; @groups.empty?; end
        # Returns this batch's groups and starts a new, empty batch.
        def drain
          groups = @groups.values
          @groups = {}
          groups
        end
      end
      # Collapses one actor's buffered events into a single event. A lone event is
      # returned unchanged, so its formatting is identical to the unbatched case.
      class Coalescer
        SEPARATOR = '; '
        def initialize(separator = SEPARATOR); @separator = separator; end
        def call(group)
          return group.first if group.length == 1
          group.first.merge(message: group.map { |event| event[:message] }.join(@separator))
        end
      end
      def initialize(sink)
        @rules = RuleChain.new([Rule.new])
        @formatter = Formatter.new
        @delivery = Delivery.new(sink)
        @buffer = Buffer.new
        @coalescer = Coalescer.new
      end
      # Adds an eligible event to the current batch. Nothing is delivered until
      # flush; ineligible events are dropped here and never reach the buffer.
      def publish(event)
        @buffer.add(event) if @rules.eligible?(event)
        nil
      end
      # Delivers the current batch as one message per actor, in first-seen actor
      # order, then starts a new batch. An empty buffer delivers nothing.
      def flush
        @buffer.drain.each do |group|
          @delivery.call(@formatter.call(@coalescer.call(group)))
        end
        nil
      end
    end
