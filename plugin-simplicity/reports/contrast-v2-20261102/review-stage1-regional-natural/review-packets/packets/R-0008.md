### Final

    module ShippingThresholds
      # Subtotal at or above which shipping is free, per warehouse region.
      #
      # Both quote paths (CheckoutShipping and DeliveryEstimate) read these, so a
      # change here applies consistently to every quote for that warehouse.
      FREE_SHIPPING = {
        north: 50.0,
        south: 75.0
      }.freeze
    
      # The warehouse contract covers north and south only. Any other region keeps
      # the historical threshold rather than inheriting south's renegotiated terms.
      DEFAULT_FREE_SHIPPING = 50.0
    
      private
    
      def free_shipping_threshold(region)
        FREE_SHIPPING.fetch(region, DEFAULT_FREE_SHIPPING)
      end
    end
    
    class CheckoutShipping
      include ShippingThresholds
    
      def quote(region:, subtotal:, weight_kg:)
        return 0.0 if subtotal >= free_shipping_threshold(region)

        else

        end
      end
    end
