      def quote(region:, subtotal:, weight_kg:)

    module FreeShipping
      # Per-warehouse free-shipping thresholds. Single source of truth: both
      # CheckoutShipping and DeliveryEstimate read from here so the two can never
      # quote different free-shipping terms for the same region.
      # Both thresholds are inclusive: a subtotal exactly at the threshold ships free.
      NORTH_THRESHOLD = 50.0
      SOUTH_THRESHOLD = 75.0
    
      def self.threshold(region)
        region == :north ? NORTH_THRESHOLD : SOUTH_THRESHOLD
      end
    end
