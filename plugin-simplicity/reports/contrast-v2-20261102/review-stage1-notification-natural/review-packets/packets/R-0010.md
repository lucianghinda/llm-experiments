## Neutral behavior contract

Users sometimes post several notifications in one burst, and a burst should appear as one message per person. Add this behavior while preserving the current formatting and delivery behavior.

      class RuleChain

      end
      class Formatter
        def call(event); "#{event[:actor]}: #{event[:message]}"; end
      end

### Final

    class NotificationPipeline
      class Rule; def call(event); !event[:system]; end; end
      class RuleChain
        def initialize(rules); @rules = rules; end
        def eligible?(event); @rules.all? { |rule| rule.call(event) }; end
      end
      class Formatter
        SEPARATOR = '; '
        def call(event); "#{event[:actor]}: #{event[:message]}"; end
        def coalesce(events)
          call(actor: events.first[:actor], message: events.map { |event| event[:message] }.join(SEPARATOR))
        end
      end
      class Delivery
        def initialize(sink); @sink = sink; end
        def call(text); @sink << text; end
      end
      def initialize(sink)
        @rules = RuleChain.new([Rule.new])
        @formatter = Formatter.new
        @delivery = Delivery.new(sink)
      end
      def publish(event)
        publish_burst([event])
      end
      # Notifications posted together are one message per actor, in the order each actor first appears.
      def publish_burst(events)
        events.select { |event| @rules.eligible?(event) }
              .group_by { |event| event[:actor] }
              .each_value { |burst| @delivery.call(@formatter.coalesce(burst)) }
      end
    end
